ขั้นตอนการตั้งค่า Backend (ทำเพียงครั้งเดียว):

  1. ไปที่ Google Apps Script แล้วสร้างโปรเจกต์ใหม่
  2. คัดลอกซอร์สโค้ดในกล่องด้านล่างนี้ไปวางในไฟล์ Code.gs
  3. กดปุ่ม **Deploy (การทำให้ใช้งานได้)** → **New deployment (การทำให้ใช้งานได้ใหม่)**
  4. เลือกประเภทเป็น **Web App (แอปเว็บ)**
  5. ตั้งค่า Access: **"Anyone" (ทุกคน)** แล้วกด Deploy
  6. คัดลอก **Web App URL** มาวางในช่องด้านบนแล้วกด "บันทึกการเชื่อมต่อ"
โค้ด Google Apps Script (Code.gs): 
// ==========================================
// Google Apps Script Backend for Smart Homework
// ==========================================

const SHEET_NAME = "Homeworks";
const DRIVE_FOLDER_NAME = "Homework_Attachments";

function doGet(e) {
  const action = e.parameter.action;
  if (action === "getHomeworks") {
    return jsonResponse(getHomeworks());
  }
  return jsonResponse({ status: "success", message: "API is ready" });
}

function doPost(e) {
  try {
    const data = JSON.parse(e.postData.contents);
    const action = data.action;

    if (action === "addHomework") {
      return jsonResponse(addHomework(data));
    } else if (action === "updateStatus") {
      return jsonResponse(updateStatus(data.id, data.status));
    } else if (action === "deleteHomework") {
      return jsonResponse(deleteHomework(data.id));
    }

    return jsonResponse({ status: "error", message: "Invalid Action" });
  } catch (err) {
    return jsonResponse({ status: "error", message: err.toString() });
  }
}

function getSheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) {
    sheet = ss.insertSheet(SHEET_NAME);
    // สร้างหัวตารางอัตโนมัติ
    sheet.appendRow(["ID", "Subject", "Title", "Description", "DueDate", "Priority", "ImageUrl", "Status", "CreatedAt"]);
    sheet.getRange(1, 1, 1, 9).setFontWeight("bold").setBackground("#4f46e5").setFontColor("#ffffff");
  }
  return sheet;
}

function getHomeworks() {
  const sheet = getSheet();
  const data = sheet.getDataRange().getValues();
  if (data.length <= 1) return [];

  const headers = data[0];
  const rows = data.slice(1);
  return rows.map(row => {
    return {
      id: row[0],
      subject: row[1],
      title: row[2],
      description: row[3],
      dueDate: row[4],
      priority: row[5],
      imageUrl: row[6],
      status: row[7],
      createdAt: row[8]
    };
  });
}

function addHomework(data) {
  const sheet = getSheet();
  let imageUrl = "";

  // จัดเก็บรูปภาพลง Google Drive (ถ้ามีการแนบไฟล์มา)
  if (data.fileBase64 && data.fileName) {
    let folder;
    const folders = DriveApp.getFoldersByName(DRIVE_FOLDER_NAME);
    if (folders.hasNext()) {
      folder = folders.next();
    } else {
      folder = DriveApp.createFolder(DRIVE_FOLDER_NAME);
    }
    
    const splitData = data.fileBase64.split(',');
    const contentType = splitData[0].match(/:(.*?);/)[1];
    const bytes = Utilities.base64Decode(splitData[1]);
    const blob = Utilities.newBlob(bytes, contentType, data.fileName);
    const file = folder.createFile(blob);
    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    imageUrl = "https://lh3.googleusercontent.com/d/" + file.getId();
  }

  const newId = "HW_" + new Date().getTime();
  const createdAt = new Date().toISOString();

  sheet.appendRow([
    newId,
    data.subject,
    data.title,
    data.description || "",
    data.dueDate,
    data.priority,
    imageUrl,
    "Pending",
    createdAt
  ]);

  return { status: "success", id: newId, imageUrl: imageUrl };
}

function updateStatus(id, newStatus) {
  const sheet = getSheet();
  const data = sheet.getDataRange().getValues();
  for (let i = 1; i < data.length; i++) {
    if (data[i][0] == id) {
      sheet.getRange(i + 1, 8).setValue(newStatus);
      return { status: "success" };
    }
  }
  return { status: "error", message: "Not found" };
}

function deleteHomework(id) {
  const sheet = getSheet();
  const data = sheet.getDataRange().getValues();
  for (let i = 1; i < data.length; i++) {
    if (data[i][0] == id) {
      sheet.deleteRow(i + 1);
      return { status: "success" };
    }
  }
  return { status: "error", message: "Not found" };
}

function jsonResponse(data) {
  return ContentService.createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}
